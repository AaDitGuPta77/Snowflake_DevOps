CREATE OR REPLACE TABLE tbl_orders (
    order_id NUMBER(38,0) AUTOINCREMENT,
    customer_name STRING,
    order_amount NUMBER(10,2),
    order_status STRING,
    priority_level STRING DEFAULT 'STANDARD',
    created_at TIMESTAMP_NTZ DEFAULT CURRENT_TIMESTAMP()
);

CREATE OR REPLACE TABLE tbl_customers (
    customer_id NUMBER(38,0) AUTOINCREMENT,
    first_name STRING,
    last_name STRING,
    email STRING,
    account_status STRING DEFAULT 'ACTIVE',
    created_at TIMESTAMP_NTZ DEFAULT CURRENT_TIMESTAMP()
);
